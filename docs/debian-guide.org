** 🐙 My previous work setup: Debian Guide

This was my daily driver for work machines.

- Go ahead and download Debian GNOME latest ISO
- dd to a USB drive, boot from that USB drive on your computer
- Install Debian to your hard-drive (or portable device)
- Use LVM + encryption option is recommended (ext4) or other advanced partitioning
- Upon login for the 1st time, you should enable non-free firmware and configure the correct time-zone
- Change your apt sources to point to ~unstable~ (e.g. ~/etc/apt/sources.list.d/heks.sources~)

#+begin_example
Types: deb
URIs: https://deb.debian.org/debian/
Suites: sid
Components: main contrib non-free non-free-firmware
Signed-By: /usr/share/keyrings/debian-archive-keyring.gpg
#+end_example

- Reboot, then run apt update and apt full-upgrade. You may also wanna ~apt install firmware-linux-nonfree uidmap~

For Niri on Debian, an easy way is to use pacstall (the AUR for deb-based distros)

#+begin_src shell
  sudo bash -c "$(curl -fsSL https://pacstall.dev/q/install || wget -q https://pacstall.dev/q/install -O -)"
  pacstall -I niri
#+end_src

#+begin_src shell
curl -fsSL https://dl.k6.io/key.gpg \
| sudo gpg --dearmor -o /usr/share/keyrings/k6-archive-keyring.gpg

gpg --no-default-keyring \
  --keyring /usr/share/keyrings/k6-archive-keyring.gpg \
  --list-keys

echo "deb [signed-by=/usr/share/keyrings/k6-archive-keyring.gpg] https://dl.k6.io/deb stable main" \
| sudo tee /etc/apt/sources.list.d/k6.list

sudo apt update
sudo apt install k6
#+end_src

(define (install-apt)
  "Install Heks Linux system APT packages (DEB Debian, Ubuntu, etc)."
  (time-machine-manifest-shell (list "guile -L ./src -c"
                                "'((@(heks-linux packages apt) install-apt))'")
                               #:verbose? #t))

(define (update-apt)
  "Update APT package manager."
  ($ '("sudo apt update -y")
     #:verbose? #t)
  ($ '("sudo apt upgrade -y")
     #:verbose? #t))
#+end_src

(define heks-linux-apt
  (let* ((xs (delete-duplicates (append (heks-linux-cli-apt)
                                        (heks-linux-desktop-apt)
                                        (heks-linux-dev-apt)
                                        (heks-linux-dict-apt)
                                        (heks-linux-font-apt)
                                        (heks-linux-graphics-apt)
                                        (heks-linux-lib-apt)
                                        (heks-linux-media-apt)
                                        (heks-linux-music-apt)
                                        (heks-linux-net-apt)
                                        (heks-linux-other-apt)
                                        (heks-linux-power-user-apt)
                                        (heks-linux-printing-apt)))))
    (sort xs
          (lambda (a b)
            (string<? (if (symbol? a)
                          (symbol->string a) a)
                      (if (symbol? b)
                          (symbol->string b) b))))))

(define (install-apt)
  (let* ((cmd (format #f "sudo apt-get install -y ~a"
                      (string-join (map (lambda (s)
                                          (format #f "~a" s)) heks-linux-apt)
                                   " "))))
    (format #t "\nExecuting command: ~a\n" cmd)
    (syscall cmd)))
